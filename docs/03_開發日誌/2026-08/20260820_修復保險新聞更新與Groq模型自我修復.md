  ```typescript
  // 核心動態自我修復邏輯 (自 Groq 在線 API 探索最新 LLM 模型)
  const modelsRes = await fetch('https://api.groq.com/openai/v1/models', {
    headers: { 'Authorization': `Bearer ${GROQ_API_KEY.trim()}` },
  });
  if (modelsRes.ok) {
    const liveModels = (await modelsRes.json()).data || [];
    const discoveredModels = liveModels
      .map((m: any) => m.id as string)
      .filter(id => id && !triedModels.has(id) && !isNonTextModel(id));
    for (const model of discoveredModels) {
      const output = await tryModel(model);
      if (output) return output; // 自我修復成功！
    }
  }
  ```
